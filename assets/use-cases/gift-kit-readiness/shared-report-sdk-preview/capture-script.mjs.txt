async function downloaded(page) {

const fill = (page, label, value) => page.getByLabel(label, { exact: true }).fill(String(value));
async function scope(page, prefix, values) {
  const names = { batchId: 'Campaign', insertRevision: 'Insert revision', destinationRevision: 'Destination revision', checkoutRevision: 'Checkout revision', deviceScope: 'Test device', accountScope: 'Intended account' };

async function record(page, { kind, reference, minute, recorded = minute, related, amount, duration, quantity, started, result = 'yes' }) {
  await page.getByLabel('Operating event', { exact: true }).selectOption(kind);
  await fill(page, 'External record reference', reference);
  await fill(page, 'External record source', `Fictional browser fixture record ${reference}`);
  await fill(page, 'Event time (elapsed minute)', minute); await fill(page, 'Recorded time (elapsed minute)', recorded);
  await page.getByLabel('Actual recorded result', { exact: true }).selectOption(result);
  if (amount !== undefined) await fill(page, 'Authorized amount (USD)', amount);
  if (duration !== undefined) await fill(page, 'Actual rehearsal duration (minutes)', duration);
  if (quantity !== undefined) await fill(page, 'Actual quantity stated in this record (kits)', quantity);
  if (started !== undefined) await fill(page, 'Actual correction start (elapsed minute)', started);
  if (related) await fill(page, kind === 'rehearsal' ? 'Correction completion reference' : 'Passed rehearsal reference', related);
}
async function capture(page, key, selector = reportHostSelector) {
  const focused = selector === reportHostSelector || selector === '.results';
  if (focused) await page.getByRole('button', { name: 'Reports', exact: true }).click();
  const run = await downloaded(page), view = inspectGiftingJourney({ lab, run, config });
  const native = `${key}.workflow.json`, full = `${key}.output.json`, png = `${key}.png`;
  await fs.writeFile(path.join(output, native), lab.sources.renderJSON(run, { maxBytes: 50000000 }));
  await fs.writeFile(path.join(output, full), lab.sources.renderJSON(view.output, { maxBytes: 50000000 }));
  await page.locator(selector).screenshot({ path: path.join(output, png), animations: 'disabled' });
  const bytes = await fs.readFile(path.join(output, png));
  captures.push({ key, image: png, width: bytes.readUInt32BE(16), height: bytes.readUInt32BE(20), sha256: sha256(bytes),
    workflow: native, workflowSHA256: sha256(await fs.readFile(path.join(output, native))), output: full,
    workflowHash: run.contentHash, definitionHash: run.definition.contentHash, caseId: run.case.gifting.id,
    evaluations: run.evaluations.length, reviews: run.reviews.length,
    preferred: view.review?.preferred.map(row => ({ id: row.id, packageId: row.packageId, expenseMinor: row.expense, finishMinute: row.finishMinute })),
    completed: view.closure?.completed ?? false });
  if (focused) await page.getByRole('button', { name: 'Both', exact: true }).click();
  return run;
}
async function layout(page, width, height, key) {
  await page.setViewportSize({ width, height });
  const size = await page.evaluate(() => ({ width: document.documentElement.scrollWidth, viewport: innerWidth,
    nested: [...document.querySelectorAll('main,section,article,aside')].filter(element => ['auto', 'scroll'].includes(getComputedStyle(element).overflowY) && element.scrollHeight > element.clientHeight + 2).map(element => element.className) }));
  assert.ok(size.width <= size.viewport + 1, `Page overflow: ${JSON.stringify(size)}`); assert.deepEqual(size.nested, []);
  await page.screenshot({ path: path.join(output, `${key}.png`), fullPage: true, animations: 'disabled' });
  checks.push({ key, ...size });
}
