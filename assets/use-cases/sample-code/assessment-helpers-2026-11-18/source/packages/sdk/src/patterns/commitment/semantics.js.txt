import { COMMITMENT_DEFINITION } from './definition.js';
import { COMMITMENT_ASSESSMENT_STATUS, COMMITMENT_FINDING } from './status.js';

const record = value => value !== null && typeof value === 'object' && !Array.isArray(value);
const read = (value, path) => path.reduce((current, key) => record(current) ? current[key] : undefined, value);
const nativeHeaders = [
  [['format'], 'complete-a-commitment.assessment/1', 'COMMITMENT_ASSESSMENT_FORMAT_INVALID', 'Resolve a native commitment assessment before inspecting its claims.'],
  [['pattern'], 'complete-a-commitment/1', 'COMMITMENT_ASSESSMENT_PATTERN_INVALID', 'The assessment must belong to Complete a Commitment.'],
  [['evaluator'], COMMITMENT_DEFINITION.evaluator, 'COMMITMENT_ASSESSMENT_EVALUATOR_INVALID', 'The assessment must identify the supported commitment evaluator.'],
];
const completenessChecks = [
  ...nativeHeaders,
  [['status'], COMMITMENT_ASSESSMENT_STATUS.COMPLETE, 'COMMITMENT_ASSESSMENT_INCOMPLETE', 'The assessment must report a complete calculation.'],
  [['admission', 'valid'], true, 'COMMITMENT_ADMISSION_INVALID', 'The supplied inputs must have passed admission.'],
  [['admission', 'complete'], true, 'COMMITMENT_ADMISSION_INCOMPLETE', 'Input admission must be complete.'],
  [['coverage', 'complete'], true, 'COMMITMENT_COVERAGE_INCOMPLETE', 'The assessment must have complete declared coverage.'],
  [['coverage', 'admissionComplete'], true, 'COMMITMENT_ADMISSION_COVERAGE_INCOMPLETE', 'The retained coverage must confirm completed admission.'],
];
const native = assessment => nativeHeaders.every(([path, expected]) => read(assessment, path) === expected);

/**
 * Inspect declared native-assessment completion, not artifact validity or business
 * feasibility. Resolve report references with resolveCommitmentAssessment first.
 * These bounded checks read evaluated or validated SDK records; they do not
 * validate witness proofs, authenticate facts, compare preferences, or run rules.
 */
export function inspectAssessmentCompleteness(assessment) {
  const issues = completenessChecks.flatMap(([path, expected, code, message]) => read(assessment, path) === expected ? []
    : [Object.freeze({ code, path: `assessment.${path.join('.')}`, message })]);
  return Object.freeze({ complete: issues.length === 0, issues: Object.freeze(issues) });
}

/** Complete admitted coverage can establish that every listed plan is blocked. */
export function isAssessmentComplete(assessment) {
  return inspectAssessmentCompleteness(assessment).complete;
}

function feasibleWitnessIds(assessment) {
  if (!native(assessment) || !record(assessment.admission) || assessment.admission.valid !== true
    || ![COMMITMENT_ASSESSMENT_STATUS.COMPLETE, COMMITMENT_ASSESSMENT_STATUS.LIMITED].includes(assessment.status)
    || assessment.finding !== COMMITMENT_FINDING.FEASIBLE_WITNESS
    || !Array.isArray(assessment.feasiblePlanIds) || !assessment.feasiblePlanIds.length || !Array.isArray(assessment.plans)) return null;
  const claimed = new Set(assessment.feasiblePlanIds);
  if (claimed.size !== assessment.feasiblePlanIds.length || [...claimed].some(id => typeof id !== 'string' || !id)) return null;
  const checked = new Set(), seen = new Set();
  // Index retained identities once; large assessments must not require a search
  // through all plans for each claimed or preferred identity.
  for (const plan of assessment.plans) {
    if (!record(plan) || typeof plan.id !== 'string' || !plan.id || seen.has(plan.id)) return null;
    seen.add(plan.id);
    if (plan.complete === true && plan.feasible === true) checked.add(plan.id);
  }
  return [...claimed].every(id => checked.has(id)) ? claimed : null;
}

/** A fully checked witness can survive a limited search without establishing an optimum. */
export function hasFeasibleWitness(assessment) {
  return feasibleWitnessIds(assessment) !== null;
}

/** Established preference retains every tied preferred identity; it need not be unique. */
export function hasEstablishedPreference(assessment) {
  if (!isAssessmentComplete(assessment) || !record(assessment.preference) || assessment.preference.optimalityEstablished !== true
    || !record(assessment.recommendation) || assessment.recommendation.preferenceEstablished !== true) return false;
  const feasible = feasibleWitnessIds(assessment), recommendation = assessment.recommendation;
  if (!feasible || !Array.isArray(recommendation.tiedPlanIds) || !recommendation.tiedPlanIds.length) return false;
  const preferred = new Set(recommendation.tiedPlanIds);
  return preferred.size === recommendation.tiedPlanIds.length && [...preferred].every(id => feasible.has(id))
    && preferred.has(recommendation.planId) && recommendation.unique === (preferred.size === 1);
}
